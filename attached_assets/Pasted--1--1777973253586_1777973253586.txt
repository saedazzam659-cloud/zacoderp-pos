🧩 1. الهدف

⚙️ 4. منطق إنشاء الشركة

function createCompany(data):

    company = insertCompany(data)

    if data.period_option == "AUTO":
        period = createFiscalPeriod({
            company_id: company.id,
            start_date: data.start_date,
            end_date: endOfYear(data.start_date),
            status: "Open",
            is_default: true
        })

    if data.period_option == "CUSTOM":
        period = createFiscalPeriod({
            company_id: company.id,
            start_date: data.custom_start,
            end_date: data.custom_end,
            status: "Open",
            is_default: true
        })

    if data.period_option == "SELECT_EXISTING":
        assignPeriodToCompany(data.period_id, company.id)

    return company

🔍 6. التحقق قبل الإقفال

function validatePeriod(periodId):

    if existsDraftEntries(periodId):
        return "❌ قيود غير مرحلة"

    if existsUnbalancedEntries(periodId):
        return "❌ قيود غير متوازنة"

    if adjustmentsMissing(periodId):
        return "⚠️ تسويات غير مكتملة"

    return "OK"

function generateAdjustments():

    for each adjustment:

        if type == "Prepaid":
            createEntry(Expense → Prepaid)

        if type == "Accrued":
            createEntry(Expense → Accrued)

🔴 8. تصفير الإيرادات والمصروفات

function closePL(periodId):

    revenue = sum(Revenue)
    expenses = sum(Expenses)

    createEntry(Revenue → ProfitLoss)
    createEntry(ProfitLoss → Expenses)

    return revenue - expenses

💰 9. ترحيل الأرباح

function transferProfit(net):

    if net > 0:
        createEntry(ProfitLoss → RetainedEarnings)
    else:
        createEntry(RetainedEarnings → ProfitLoss)

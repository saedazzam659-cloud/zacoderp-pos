// ===============================
// جلب السياسة (افتراضي السعودية)
// ===============================
function getCountryPolicy(countryCode, lang = "ar") {

  if(!countryCode){
    countryCode = "SA";
  }

  const country = countries.find(c => c.code === countryCode);

  if(!country){
    return lang === "en" ? countries[0].policy_en : countries[0].policy_ar;
  }

  return lang === "en" ? country.policy_en : country.policy_ar;
}

// ===============================
// عند تغيير الدولة
// ===============================
function onCountryChange(countryCode, lang = "ar") {

  const country = countries.find(c => c.code === countryCode) || countries[0];

  setRegistrationPolicy(
    lang === "en" ? country.policy_en : country.policy_ar
  );

  updateCompanyForm({
    country: country.code,
    currency: country.currency
  });
}

// ===============================
// التحقق من التسجيل
// ===============================
function validateRegistration(acceptPolicy){

  if(!acceptPolicy){
    throw "يجب الموافقة على سياسة الدولة / You must accept the country policy";
  }

  return true;
}

// ===============================
// التهيئة الافتراضية
// ===============================
function initRegistration(lang = "ar"){

  const defaultCountry = "SA";

  const country = countries.find(c => c.code === defaultCountry);

  setRegistrationPolicy(
    lang === "en" ? country.policy_en : country.policy_ar
  );

  updateCompanyForm({
    country: defaultCountry,
    currency: "SAR"
  });
}
